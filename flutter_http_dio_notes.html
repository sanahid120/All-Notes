JSON & dart:convert

Flutter uses dart:convert to encode/decode JSON. Always import it at the top of any file that touches network data.

json_basics.dart
import 'dart:convert';

// ── Decode JSON string → Dart Map ──
String jsonStr = '{"name":"Rahim","age":25}';
Map<String, dynamic> data = jsonDecode(jsonStr);
print(data['name']); // Rahim

// ── Encode Dart Map → JSON string ──
Map<String, dynamic> map = {'name': 'Rahim', 'age': 25};
String encoded = jsonEncode(map);
print(encoded); // {"name":"Rahim","age":25}

// ── Decode JSON array ──
String jsonArray = '[{"id":1},{"id":2}]';
List<dynamic> list = jsonDecode(jsonArray);
print(list.length); // 2

Model Classes — fromJson / toJson

Every API response must be mapped to a Dart model class. Two key factory methods: fromJson (decode) and toJson (encode).

user_model.dart
class UserModel {
  final String id;
  final String email;
  final String firstName;
  final String lastName;
  final String mobile;

  // ── Computed property ──
  String get fullName => "$firstName $lastName";

  UserModel({
    required this.id,
    required this.email,
    required this.firstName,
    required this.lastName,
    required this.mobile,
  });

  // ── Decode API response → UserModel ──
  factory UserModel.fromJson(Map<String, dynamic> json) {
    return UserModel(
      id:        json['_id'],
      email:     json['email'],
      firstName: json['firstName'],
      lastName:  json['lastName'],
      mobile:    json['mobile'],
    );
  }

  // ── Encode UserModel → Map (for POST/PUT) ──
  Map<String, dynamic> toJson() => {
    '_id':       id,
    'email':     email,
    'firstName': firstName,
    'lastName':  lastName,
    'mobile':    mobile,
  };
}

class TaskModel {

  final String? title;
  final String? description;

  final String? email;
  final String? createdDate;

    required this.sId,
    required this.title,
    required this.description,
    required this.status,
    required this.email,
    required this.createdDate,
  });

  factory TaskModel.fromJson(Map<String, dynamic> json) {
    return TaskModel(
      sId:         json['_id'],
      title:       json['title'],
      description: json['description'],
      status:      json['status'],
      email:       json['email'],

      createdDate: DateFormat('dd MMM yyyy').format(

    );
  }

  Map<String, dynamic> toJson() => {
    '_id':         sId,
    'title':       title,
    'description': description,
    'status':      status,
    'email':       email,
    'createdDate': createdDate,
  };

Core NetworkCaller Class (HTTP Package)

Instead of writing try/catch in every screen, we create a single NetworkCaller class that wraps all HTTP logic. This is the same pattern used in your uploaded network_caller.dart, extended here.

core/network/network_response.dart
class NetworkResponse {
  final bool    isSuccess;
  final int     responseCode;
  final dynamic body;
  final String? errorMessage;

  NetworkResponse({
    required this.isSuccess,
    required this.responseCode,
    this.body,
    this.errorMessage,
  });
}
core/network/network_caller.dart
import 'dart:convert';
import 'package:flutter/material.dart';
import 'package:http/http.dart' as http;

class NetworkCaller {
  // ── Base URL ──
  static const String _baseUrl = 'https://api.example.com';

  // ── Token getter (set from AuthController) ──
  static String? accessToken;

  // ──────────────── GET ────────────────
  static Future<NetworkResponse> getRequest(String path) async {
    try {
      final uri = Uri.parse('$_baseUrl$path');
      _log('GET', path);

      final response = await http.get(uri,
        headers: _headers());

      return _handleResponse(response);
    } catch (e) {
      return _errorResponse(e);
    }
  }

  // ──────────────── POST ────────────────
  static Future<NetworkResponse> postRequest(
    String path, {
    Map<String, dynamic>? body,
  }) async {
    try {
      final uri = Uri.parse('$_baseUrl$path');
      _log('POST', path, body: body);

      final response = await http.post(uri,
        headers: _headers(withContentType: true),
        body: jsonEncode(body));

      return _handleResponse(response);
    } catch (e) {
      return _errorResponse(e);
    }
  }

  // ──────────────── PUT ────────────────
  static Future<NetworkResponse> putRequest(
    String path, {
    Map<String, dynamic>? body,
  }) async {
    try {
      final uri = Uri.parse('$_baseUrl$path');
      _log('PUT', path, body: body);

      final response = await http.put(uri,
        headers: _headers(withContentType: true),
        body: jsonEncode(body));

      return _handleResponse(response);
    } catch (e) {
      return _errorResponse(e);
    }
  }

  // ──────────────── PATCH ────────────────
  static Future<NetworkResponse> patchRequest(
    String path, {
    Map<String, dynamic>? body,
  }) async {
    try {
      final uri = Uri.parse('$_baseUrl$path');
      _log('PATCH', path, body: body);

      final response = await http.patch(uri,
        headers: _headers(withContentType: true),
        body: jsonEncode(body));

      return _handleResponse(response);
    } catch (e) {
      return _errorResponse(e);
    }
  }

  // ──────────────── DELETE ────────────────
  static Future<NetworkResponse> deleteRequest(String path) async {
    try {
      final uri = Uri.parse('$_baseUrl$path');
      _log('DELETE', path);

      final response = await http.delete(uri,
        headers: _headers());

      return _handleResponse(response);
    } catch (e) {
      return _errorResponse(e);
    }
  }

  // ─── Private Helpers ───

  static Map<String, String> _headers({bool withContentType = false}) {
    return {
      if (withContentType) 'Content-Type': 'application/json',
      'token': accessToken ?? '',
    };
  }

  static NetworkResponse _handleResponse(http.Response response) {
    final decoded = jsonDecode(response.body);

    if (response.statusCode == 200) {
      return NetworkResponse(isSuccess: true, responseCode: 200, body: decoded);
    } else if (response.statusCode == 401) {
      _onUnauthorized();
      return NetworkResponse(isSuccess: false, responseCode: 401,
                              errorMessage: 'Token expired');
    } else {
      return NetworkResponse(isSuccess: false,
                              responseCode: response.statusCode,
                              errorMessage: decoded['message'] ?? 'Unknown error');
    }
  }

  static NetworkResponse _errorResponse(dynamic e) {
    return NetworkResponse(isSuccess: false, responseCode: -1,
                            errorMessage: e.toString());
  }

  static void _onUnauthorized() {
    // Navigate to login screen and clear token
    accessToken = null;
  }

  static void _log(String method, String path,
      {Map? body}) {
    debugPrint('[NET] $method $path | body: $body');
  }
}

Dio — Setup & BaseOptions

Dio is a powerful HTTP client. Unlike the http package, it supports interceptors, global timeout, automatic JSON decoding, FormData, and download progress.

core/network/dio_client.dart
import 'package:dio/dio.dart';

class DioClient {
  static final Dio _dio = Dio(
    BaseOptions(
      baseUrl:        'https://api.example.com',
      connectTimeout: const Duration(seconds: 10),
      receiveTimeout: const Duration(seconds: 15),
      headers: {
        'Content-Type': 'application/json',
      },
      // Dio decodes JSON automatically
      responseType: ResponseType.json,
    ),
  );

  // ── Expose Dio instance ──
  static Dio get instance => _dio;

  // ── Set auth token (call after login) ──
  static void setToken(String token) {
    _dio.options.headers['token'] = token;
  }

  // ── Clear token (call after logout) ──
  static void clearToken() {
    _dio.options.headers.remove('token');
  }
}

Dio Interceptors — Token, Logging, Retry

Interceptors run automatically before every request and after every response — perfect for adding auth tokens, logging, and handling 401s globally.

core/network/auth_interceptor.dart
class AuthInterceptor extends Interceptor {

  // ── Before request: inject token ──
  @override
  void onRequest(RequestOptions options, RequestInterceptorHandler handler) {
    final token = AuthStorage.getToken();
    if (token != null) {
      options.headers['token'] = token;
    }
    handler.next(options); // continue
  }

  // ── After response ──
  @override
  void onResponse(Response response, ResponseInterceptorHandler handler) {
    debugPrint('[✅ ${response.statusCode}] ${response.requestOptions.path}');
    handler.next(response);
  }

  // ── On error: handle 401 globally ──
  @override
  void onError(DioException err, ErrorInterceptorHandler handler) {
    if (err.response?.statusCode == 401) {
      // Token expired → logout
      AuthStorage.clearToken();
      // Navigate to login
      NavigationService.goToLogin();
    }
    handler.next(err);
  }
}

// ── Register in DioClient ──
class DioClient {
  static final Dio _dio = Dio(BaseOptions(baseUrl: 'https://api.example.com'))
    ..interceptors.addAll([
      AuthInterceptor(),
      LogInterceptor(requestBody: true, responseBody: true),
    ]);

  static Dio get instance => _dio;
}

Dio Error Handling — DioException Types

dio_error_handling.dart
Future<void> safeApiCall() async {
  try {
    final response = await DioClient.instance.get('/tasks');
    print(response.data);
  } on DioException catch (e) {
    switch (e.type) {
      case DioExceptionType.connectionTimeout:
        print('⏱️ Connection timed out');
        break;
      case DioExceptionType.receiveTimeout:
        print('⏱️ Server too slow');
        break;
      case DioExceptionType.badResponse:
        final code = e.response?.statusCode;
        print('🔴 Server error: $code');
        break;
      case DioExceptionType.connectionError:
        print('📡 No internet connection');
        break;
      case DioExceptionType.cancel:
        print('❌ Request cancelled');
        break;
      default:
        print('Unknown error: ${e.message}');
    }
  }
}

Authentication — Login, Token Storage, Logout

data/services/auth_service.dart
import 'package:shared_preferences/shared_preferences.dart';

class AuthStorage {
  static const String _tokenKey = 'access_token';
  static const String _userKey  = 'user_data';

  // ── Save token after login ──
  static Future<void> saveToken(String token) async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString(_tokenKey, token);
    DioClient.setToken(token); // update Dio header too
  }

  // ── Read token ──
  static Future<String?> getToken() async {
    final prefs = await SharedPreferences.getInstance();
    return prefs.getString(_tokenKey);
  }

  // ── Save user object ──
  static Future<void> saveUser(UserModel user) async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString(_userKey, jsonEncode(user.toJson()));
  }

  // ── Logout: clear everything ──
  static Future<void> logOut() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.clear();
    DioClient.clearToken();
  }
}

// ── Login API call ──
class AuthRepository {
  Future<UserModel?> login(String email, String password) async {
    final response = await NetworkCaller.postRequest(
      '/login',
      body: {'email': email, 'password': password},
    );

    if (response.isSuccess) {
      final token = response.body['token'];
      final user  = UserModel.fromJson(response.body['data']);
      await AuthStorage.saveToken(token);
      await AuthStorage.saveUser(user);
      return user;
    }
    return null;
  }
}

File Upload with Dio — MultipartFile & FormData

file_upload.dart
import 'package:image_picker/image_picker.dart';

Future<void> uploadProfilePhoto() async {
  // 1️⃣ Pick image from gallery
  final picker = ImagePicker();
  final picked = await picker.pickImage(source: ImageSource.gallery);
  if (picked == null) return;

  // 2️⃣ Build FormData
  final formData = FormData.fromMap({
    'photo': await MultipartFile.fromFile(
      picked.path,
      filename: 'photo.jpg',
    ),
    'email': 'user@example.com', // extra fields
  });

  // 3️⃣ Send with Dio
  try {
    final response = await DioClient.instance.patch(
      '/profile/photo',
      data: formData,
      onSendProgress: (sent, total) {
        final percent = (sent / total * 100).toStringAsFixed(0);
        print('Upload: $percent%');
      },
    );
    print(response.data);
  } on DioException catch (e) {
    print('Upload failed: ${e.message}');
  }
}
⚠️
Do NOT set Content-Type: application/json when sending FormData. Dio sets the multipart boundary automatically.

Repository Pattern — Separating API from UI

The repository sits between the UI/controllers and the network layer. UI never calls NetworkCaller directly.

data/repositories/task_repository.dart
class TaskRepository {
  // ── Get all tasks ──
  Future<List<TaskModel>?> getAllTasks() async {
    final res = await NetworkCaller.getRequest('/tasks');
    if (res.isSuccess) {
      final rawList = res.body['data'] as List;
      return rawList.map((e) => TaskModel.fromJson(e)).toList();
    }
    return null;
  }

  // ── Get tasks by status ──
  Future<List<TaskModel>?> getTasksByStatus(String status) async {
    final res = await NetworkCaller.getRequest(
      '/tasks?status=$status');
    if (res.isSuccess) {
      final rawList = res.body['data'] as List;
      return rawList.map((e) => TaskModel.fromJson(e)).toList();
    }
    return null;
  }

  // ── Create task ──
  Future<bool> createTask(String title, String desc) async {
    final res = await NetworkCaller.postRequest(
      '/tasks',
      body: {'title': title, 'description': desc, 'status': 'New'},
    );
    return res.isSuccess;
  }

  // ── Update status (PATCH) ──
  Future<bool> updateStatus(String id, String newStatus) async {
    final res = await NetworkCaller.patchRequest(
      '/tasks/$id',
      body: {'status': newStatus},
    );
    return res.isSuccess;
  }

  // ── Delete task ──
  Future<bool> deleteTask(String id) async {
    final res = await NetworkCaller.deleteRequest('/tasks/$id');
    return res.isSuccess;
  }
}

Complete Working Example — Task Screen

Putting it all together: a task screen that loads, creates, updates, and deletes tasks.

presentation/screens/task_screen.dart
class TaskScreen extends StatefulWidget {
  const TaskScreen({super.key});
  @override
  State<TaskScreen> createState() => _TaskScreenState();
}

class _TaskScreenState extends State<TaskScreen> {
  final _repo = TaskRepository();
  List<TaskModel> _tasks = [];
  bool _loading = false;
  String? _error;

  @override
  void initState() {
    super.initState();
    _loadTasks();
  }

  // ── Load tasks ──
  Future<void> _loadTasks() async {
    setState(() => _loading = true);
    final tasks = await _repo.getAllTasks();
    setState(() {
      _loading = false;
      if (tasks != null) {
        _tasks = tasks;
      } else {
        _error = 'Failed to load tasks';
      }
    });
  }

  // ── Create task ──
  Future<void> _createTask(String title, String desc) async {
    final success = await _repo.createTask(title, desc);
    if (success) _loadTasks(); // refresh
  }

  // ── Update status ──
  Future<void> _updateStatus(String id, String status) async {
    final ok = await _repo.updateStatus(id, status);
    if (ok) _loadTasks();
  }

  // ── Delete task ──
  Future<void> _deleteTask(String id) async {
    final ok = await _repo.deleteTask(id);
    if (ok) _loadTasks();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Tasks')),
      body: _loading
        ? const Center(child: CircularProgressIndicator())
        : _error != null
          ? Center(child: Text(_error!))
          : ListView.builder(
              itemCount: _tasks.length,
              itemBuilder: (ctx, i) {
                final task = _tasks[i];
                return ListTile(
                  title:    Text(task.title ?? ''),
                  subtitle: Text(task.status ?? ''),
                  trailing: Row(
                    mainAxisSize: MainAxisSize.min,
                    children: [
                      IconButton(
                        icon: const Icon(Icons.check),
                        onPressed: () =>
                          _updateStatus(task.sId!, 'Completed'),
                      ),
                      IconButton(
                        icon: const Icon(Icons.delete, color: Colors.red),
                        onPressed: () => _deleteTask(task.sId!),
                      ),
                    ],
                  ),
                );
              },
            ),
      floatingActionButton: FloatingActionButton(
        onPressed: () => _createTask('New Task', 'Description'),
        child: const Icon(Icons.add),
      ),
    );
  }

Best Practices — Do's & Don'ts

✅
DO: Use Dio in production apps — interceptors save a huge amount of boilerplate code.
✅
DO: Create a Model class for every API response. Never parse JSON directly inside UI widgets.
✅
DO: Keep all API URLs in one ApiEndpoints class so changing the backend URL is a one-line edit.
✅
DO: Handle 401 globally in an interceptor — never repeat logout logic in every screen.
❌
DON'T: Put http.get() or dio.get() calls directly in your widget's build() method.
❌
DON'T: Store secret keys or tokens in source code — use SharedPreferences, flutter_secure_storage, or .env files.
❌
DON'T: Set Content-Type: application/json when sending FormData — Dio sets the multipart header automatically.
🔀
HTTP vs Dio: Use http package for simple hobby projects. Use Dio for any real production app that needs interceptors, timeout config, upload progress, or cancellation.
