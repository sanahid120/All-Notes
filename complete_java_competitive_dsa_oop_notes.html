13. Searching and Sorting Algorithms

Linear search

static int linearSearch(int[] array, int target) {
    for (int i = 0; i < array.length; i++) {
        if (array[i] == target) return i;
    }
    return -1;
}

Binary search

static int binarySearch(int[] array, int target) {
    int left = 0, right = array.length - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (array[mid] == target) return mid;
        if (array[mid] < target) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}

Bubble sort

static void bubbleSort(int[] array) {
    for (int i = 0; i < array.length; i++) {
        for (int j = 0; j + 1 < array.length - i; j++) {
            if (array[j] > array[j + 1]) {
                int temp = array[j];
                array[j] = array[j + 1];
                array[j + 1] = temp;
            }
        }
    }
}

Merge sort

static void mergeSort(int[] a, int l, int r) {
    if (l >= r) return;
    int m = l + (r - l) / 2;
    mergeSort(a, l, m);
    mergeSort(a, m + 1, r);
    merge(a, l, m, r);
}

static void merge(int[] a, int l, int m, int r) {
    int[] temp = new int[r - l + 1];
    int i = l, j = m + 1, k = 0;
    while (i <= m && j <= r) temp[k++] = a[i] <= a[j] ? a[i++] : a[j++];
    while (i <= m) temp[k++] = a[i++];
    while (j <= r) temp[k++] = a[j++];
    for (int x = 0; x < temp.length; x++) a[l + x] = temp[x];
}

static TreeNode insert(TreeNode root, int value) {
    if (root == null) return new TreeNode(value);
    if (value < root.value) root.left = insert(root.left, value);
    else root.right = insert(root.right, value);
    return root;
}

static void inorder(TreeNode root) {
    if (root == null) return;
    inorder(root.left);
    System.out.print(root.value + " ");
    inorder(root.right);
}

BFS

static void bfs(int start, List> graph) {
    boolean[] visited = new boolean[graph.size()];
    Queue queue = new ArrayDeque<>();
    visited[start] = true;
    queue.offer(start);

    while (!queue.isEmpty()) {
        int node = queue.poll();
        System.out.print(node + " ");
        for (int next : graph.get(node)) {
            if (!visited[next]) {
                visited[next] = true;
                queue.offer(next);
            }
        }
    }
}

DFS

static void dfs(int node, List> graph, boolean[] visited) {
    visited[node] = true;
    System.out.print(node + " ");
    for (int next : graph.get(node)) {
        if (!visited[next]) dfs(next, graph, visited);
    }
}

17. Recursion, Backtracking, Greedy, and Dynamic Programming

static long factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}

Backtracking subsets

static void subsets(int index, int[] a, List current) {
    if (index == a.length) {
        System.out.println(current);
        return;
    }
    subsets(index + 1, a, current);
    current.add(a[index]);
    subsets(index + 1, a, current);
    current.remove(current.size() - 1);
}

Dynamic programming

static long fibonacci(int n) {
    long[] dp = new long[n + 1];
    if (n >= 1) dp[1] = 1;
    for (int i = 2; i <= n; i++) dp[i] = dp[i - 1] + dp[i - 2];
    return dp[n];
}

19. Number Theory and Bit Manipulation

static long gcd(long a, long b) {
    while (b != 0) {
        long temp = a % b;
        a = b;
        b = temp;
    }
    return Math.abs(a);
}

static long lcm(long a, long b) {
    return Math.abs(a / gcd(a, b) * b);
}

Prime and sieve

static boolean isPrime(long n) {
    if (n < 2) return false;
    for (long i = 2; i * i <= n; i++) if (n % i == 0) return false;
    return true;
}

boolean[] prime = new boolean[n + 1];
Arrays.fill(prime, true);
prime[0] = prime[1] = false;
for (int p = 2; p * p <= n; p++) {
    if (prime[p]) {
        for (int multiple = p * p; multiple <= n; multiple += p) {
            prime[multiple] = false;
        }
    }
}

20. DSU and Dijkstra

Disjoint Set Union

class DSU {
    int[] parent, size;

    DSU(int n) {
        parent = new int[n];
        size = new int[n];
        for (int i = 0; i < n; i++) {
            parent[i] = i;
            size[i] = 1;
        }
    }

    int find(int x) {
        if (parent[x] == x) return x;
        return parent[x] = find(parent[x]);
    }

    boolean union(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return false;
        if (size[a] < size[b]) {
            int t = a; a = b; b = t;
        }
        parent[b] = a;
        size[a] += size[b];
        return true;
    }
}

Dijkstra

record Edge(int to, int weight) {}
record State(long distance, int node) implements Comparable {
    public int compareTo(State other) {
        return Long.compare(distance, other.distance);
    }
}

static long[] dijkstra(int source, List> graph) {
    long[] dist = new long[graph.size()];
    Arrays.fill(dist, Long.MAX_VALUE);
    PriorityQueue pq = new PriorityQueue<>();
    dist[source] = 0;
    pq.offer(new State(0, source));

    while (!pq.isEmpty()) {
        State current = pq.poll();
        if (current.distance() != dist[current.node()]) continue;
        for (Edge edge : graph.get(current.node())) {
            long next = current.distance() + edge.weight();
            if (next < dist[edge.to()]) {
                dist[edge.to()] = next;
                pq.offer(new State(next, edge.to()));
            }
        }
    }
    return dist;
}

22. Competitive Programming Template

import java.io.*;
import java.util.*;

public class Main {
    static class FastScanner {
        private final BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        private StringTokenizer st;
        String next() throws IOException {
            while (st == null || !st.hasMoreTokens()) st = new StringTokenizer(br.readLine());
            return st.nextToken();
        }
        int nextInt() throws IOException { return Integer.parseInt(next()); }
    }

    static void solve(FastScanner fs, StringBuilder out) throws Exception {
        int n = fs.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = fs.nextInt();
        Arrays.sort(a);
        for (int value : a) out.append(value).append(' ');
        out.append('\n');
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner();
        StringBuilder out = new StringBuilder();
        int testCases = fs.nextInt();
        while (testCases-- > 0) solve(fs, out);
        System.out.print(out);
    }
}

23. Learning and Practice Roadmap

  1. Learn syntax, input/output, types, loops, methods, arrays, and strings.
  2. Practice ArrayList, HashMap, HashSet, Queue, Deque, and PriorityQueue.
  3. Master OOP, interfaces, abstract classes, generics, and exceptions.
  4. Implement searching, sorting, linked lists, stacks, queues, trees, and graphs.
  5. Learn prefix sums, two pointers, sliding windows, binary search on answer, and greedy methods.
  6. Study recursion, backtracking, dynamic programming, DSU, shortest paths, and MST.
  7. Solve problems regularly and review time complexity and edge cases.
Practice rule: implement every algorithm from memory and test empty, duplicate, sorted, reverse-sorted, and boundary inputs.
