Basic Display Widgets
📝

Text

Displays a string of text with optional styling. The most fundamental Flutter widget.

StatelessWidgetDisplay
text_widget.dart
// ── Basic text ──
Text('Hello, Flutter!')

// ── Styled text ──
Text(
  'Styled Text',
  style: TextStyle(
    fontSize:   24,
    fontWeight: FontWeight.bold,
    color:      Colors.blue,
    letterSpacing: 1.5,
    decoration: TextDecoration.underline,
  ),
  textAlign:  TextAlign.center,
  maxLines:   2,
  overflow:   TextOverflow.ellipsis,
)

// ── RichText: multiple styles in one line ──
RichText(
  text: TextSpan(
    children: [
      TextSpan(text: 'Hello ',
        style: TextStyle(color: Colors.black, fontSize: 16)),
      TextSpan(text: 'Flutter',
        style: TextStyle(color: Colors.blue,
                          fontWeight: FontWeight.bold, fontSize: 16)),
    ],
  ),
)
💡
Use overflow: TextOverflow.ellipsis with maxLines to prevent text from overflowing its container.

Scaffold

The base structure for a Flutter screen. Provides appBar, body, drawer, bottomNavigationBar, and FloatingActionButton slots.

StatefulWidgetLayoutNavigation
scaffold_screen.dart
class HomeScreen extends StatelessWidget {
  const HomeScreen({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      // ── Top bar ──
      appBar: AppBar(
        title: const Text('My App'),
        backgroundColor: Colors.blue,
        actions: [
          IconButton(icon: const Icon(Icons.search),
            onPressed: () {}),
        ],
      ),

      // ── Side drawer ──
      drawer: Drawer(
        child: ListView(children: [
          const DrawerHeader(child: Text('Menu')),
          ListTile(title: const Text('Home'),
            onTap: () => Navigator.pop(context)),
        ]),
      ),

      // ── Main content ──
      body: const Center(child: Text('Body here')),

      // ── Floating button ──
      floatingActionButton: FloatingActionButton(
        onPressed: () {},
        child: const Icon(Icons.add),
      ),

      // ── Bottom bar ──
      bottomNavigationBar: BottomNavigationBar(
        currentIndex: 0,
        onTap: (i) {},
        items: const [
          BottomNavigationBarItem(
            icon: Icon(Icons.home), label: 'Home'),
          BottomNavigationBarItem(
            icon: Icon(Icons.person), label: 'Profile'),
        ],
      ),
    );
  }
}

Container

The Swiss-army-knife layout widget. Handles width, height, padding, margin, color, border, shadow, and shape all in one widget.

Container(
  width:   200,
  height:  100,
  margin:  const EdgeInsets.all(12),
  padding: const EdgeInsets.symmetric(
             horizontal: 16, vertical: 8),
  decoration: BoxDecoration(
    color:        Colors.blue,

        color:      Colors.black26,
        blurRadius: 8,
        offset:     const Offset(0, 4),

      colors: [Colors.blue, Colors.purple],

ListView & ListView.builder

Scrollable list of widgets. ListView.builder lazily builds only visible items — use it for dynamic lists from API data.

StatelessWidgetScroll
listview_widget.dart
// ── Static list (small fixed list) ──
ListView(
  padding: const EdgeInsets.all(12),
  children: const [
    ListTile(
      leading:  Icon(Icons.person),
      title:    Text('Rahim Ahmed'),
      subtitle: Text('Developer'),
      trailing: Icon(Icons.arrow_forward_ios),
    ),
    Divider(),
    ListTile(title: Text('Another item')),
  ],
)

// ── builder: for API / dynamic data ──
ListView.builder(
  itemCount:   tasks.length,
  itemBuilder: (context, index) {
    final task = tasks[index];
    return Card(
      margin: const EdgeInsets.symmetric(
        horizontal: 12, vertical: 6),
      child: ListTile(
        title:    Text(task.title ?? ''),
        subtitle: Text(task.status ?? ''),
        trailing: IconButton(
          icon: const Icon(Icons.delete,
            color: Colors.red),
          onPressed: () => _delete(task.sId!),
        ),
      ),
    );
  },
)

// ── separated: adds dividers automatically ──
ListView.separated(
  itemCount:     items.length,
  separatorBuilder: (_, __) => const Divider(),
  itemBuilder:   (ctx, i) => ListTile(
    title: Text(items[i])),
)
✅
Always use ListView.builder for lists from an API. It's memory-efficient and only renders visible items.

TextField

User text input. Use with TextEditingController to read, clear, and validate entered text.

StatefulWidgetInput
textfield_widget.dart
// ── In State class ──
final _emailCtrl    = TextEditingController();
final _passwordCtrl = TextEditingController();
final _formKey      = GlobalKey<FormState>();

// ── Email field ──
TextField(
  controller:  _emailCtrl,
  keyboardType: TextInputType.emailAddress,
  decoration: const InputDecoration(
    labelText:   'Email',
    hintText:    'you@example.com',
    prefixIcon:  Icon(Icons.email),
    border:      OutlineInputBorder(),
  ),
)

// ── Password field ──
TextField(
  controller:  _passwordCtrl,
  obscureText: _isHidden,
  decoration: InputDecoration(
    labelText:  'Password',
    border:     const OutlineInputBorder(),
    suffixIcon: IconButton(
      icon: Icon(_isHidden
        ? Icons.visibility_off
        : Icons.visibility),
      onPressed: () =>
        setState(() => _isHidden = !_isHidden),
    ),
  ),
)

// ── Form + TextFormField (with validation) ──
Form(
  key: _formKey,
  child: TextFormField(
    validator: (val) {
      if (val == null || val.isEmpty)
        return 'Required';
      if (!val.contains('@'))
        return 'Enter valid email';
      return null;
    },
    decoration: const InputDecoration(
      labelText: 'Email'),
  ),
)

// ── On submit: validate then read value ──
void _submit() {
  if (_formKey.currentState!.validate()) {
    final email = _emailCtrl.text.trim();
    print(email);
  }
}

Async Widgets — API Integration
🔮

FutureBuilder

Automatically builds different UI states (loading → error → data) based on the result of a Future, perfect for API calls.

StatelessWidgetAsync
future_builder.dart
class TaskListScreen extends StatelessWidget {
  final _repo = TaskRepository();

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Tasks')),
      body: FutureBuilder<List<TaskModel>?>(
        future: _repo.getAllTasks(),  // Future here
        builder: (context, snapshot) {

          // ── Loading ──
          if (snapshot.connectionState ==
              ConnectionState.waiting) {
            return const Center(
              child: CircularProgressIndicator());
          }

          // ── Error ──
          if (snapshot.hasError || snapshot.data == null) {
            return Center(
              child: Text('Error: ${snapshot.error}'));
          }

          // ── Data ──
          final tasks = snapshot.data!;
          return ListView.builder(
            itemCount:   tasks.length,
            itemBuilder: (ctx, i) =>
              ListTile(title: Text(tasks[i].title ?? '')),
          );
        },
      ),
    );
  }
}
💡
Store the Future in a variable — never create a new Future inside build() or it will re-call the API on every rebuild.
🌊

StreamBuilder

Rebuilds UI on every new event from a Stream — great for real-time data like chat, live scores, or Firebase.

StatelessWidgetAsync
stream_builder.dart
// ── Create a stream (e.g. a counter every second) ──
final Stream<int> countStream =
  Stream.periodic(const Duration(seconds: 1),
    (i) => i);

// ── Build UI from stream ──
StreamBuilder<int>(
  stream: countStream,
  builder: (context, snapshot) {
    if (!snapshot.hasData) {
      return const CircularProgressIndicator();
    }
    return Text(
      'Count: ${snapshot.data}',
      style: const TextStyle(fontSize: 32),
    );
  },
)

// ── With Firebase Firestore ──
StreamBuilder<QuerySnapshot>(
  stream: FirebaseFirestore.instance
    .collection('messages').snapshots(),
  builder: (ctx, snapshot) {
    if (!snapshot.hasData)
      return const CircularProgressIndicator();
    final docs = snapshot.data!.docs;
    return ListView.builder(
      itemCount: docs.length,
      itemBuilder: (_, i) =>
        Text(docs[i]['text']),
    );
  },
)

Navigator — push, pop, pushNamed, pushReplacement

Manages screen navigation with a stack. Push to go forward, pop to go back.

Navigation
navigation.dart
// ── Push: go to new screen ──
Navigator.push(context,
  MaterialPageRoute(
    builder: (_) => const DetailScreen()));

// ── Push with data ──
Navigator.push(context,
  MaterialPageRoute(
    builder: (_) => DetailScreen(task: task)));

// ── Pop: go back ──
Navigator.pop(context);

// ── Pop with result ──
Navigator.pop(context, 'updated');

// ── PushReplacement: replace current (login → home) ──
Navigator.pushReplacement(context,
  MaterialPageRoute(builder: (_) => const HomeScreen()));

// ── PushNamedAndRemoveUntil: clear all (logout) ──
Navigator.pushNamedAndRemoveUntil(
  context, '/login', (route) => false);

// ── Named routes setup in MaterialApp ──
MaterialApp(
  initialRoute: '/',
  routes: {
    '/':       (_) => const HomeScreen(),
    '/login':  (_) => const LoginScreen(),
    '/detail': (_) => const DetailScreen(),
  },
)
