14. Searching and Sorting

Linear search

int linearSearch(const vector<int>& a, int target) {
    for (int i = 0; i < (int)a.size(); i++)
        if (a[i] == target) return i;
    return -1;
}

Binary search

int binarySearch(const vector<int>& a, int target) {
    int left = 0, right = (int)a.size() - 1;

    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (a[mid] == target) return mid;
        if (a[mid] < target) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}

Bubble sort

void bubbleSort(vector<int>& a) {
    for (int i = 0; i < (int)a.size(); i++)
        for (int j = 0; j + 1 < (int)a.size() - i; j++)
            if (a[j] > a[j + 1])
                swap(a[j], a[j + 1]);
}

Merge sort

void mergeSort(vector<int>& a, int l, int r) {
    if (l >= r) return;

    int m = l + (r - l) / 2;
    mergeSort(a, l, m);
    mergeSort(a, m + 1, r);

    vector<int> temp;
    int i = l, j = m + 1;

    while (i <= m && j <= r)
        temp.push_back(a[i] <= a[j] ? a[i++] : a[j++]);

    while (i <= m) temp.push_back(a[i++]);
    while (j <= r) temp.push_back(a[j++]);

    for (int k = l; k <= r; k++)
        a[k] = temp[k - l];
}

15. Linked List

struct Node {
    int data;
    Node* next;

    explicit Node(int value) : data(value), next(nullptr) {}
};

class LinkedList {
    Node* head = nullptr;

public:
    void pushFront(int value) {
        Node* node = new Node(value);
        node->next = head;
        head = node;
    }

    void display() const {
        Node* current = head;
        while (current) {
            cout << current->data << " -> ";
            current = current->next;
        }
        cout << "null\n";
    }

    ~LinkedList() {
        while (head) {
            Node* next = head->next;
            delete head;
            head = next;
        }
    }
};

16. Trees and Binary Search Trees

struct TreeNode {
    int value;
    TreeNode* left;
    TreeNode* right;

    explicit TreeNode(int value)
        : value(value), left(nullptr), right(nullptr) {}
};

TreeNode* insert(TreeNode* root, int value) {
    if (!root) return new TreeNode(value);

    if (value < root->value)
        root->left = insert(root->left, value);
    else
        root->right = insert(root->right, value);

    return root;
}

void inorder(TreeNode* root) {
    if (!root) return;
    inorder(root->left);
    cout << root->value << ' ';
    inorder(root->right);
}

BFS

void bfs(int start, const vector<vector<int>>& graph) {
    vector<bool> visited(graph.size(), false);
    queue<int> q;

    visited[start] = true;
    q.push(start);

    while (!q.empty()) {
        int node = q.front();
        q.pop();

        cout << node << ' ';

        for (int next : graph[node]) {
            if (!visited[next]) {
                visited[next] = true;
                q.push(next);
            }
        }
    }
}

DFS

void dfs(int node, const vector<vector<int>>& graph,
         vector<bool>& visited) {
    visited[node] = true;
    cout << node << ' ';

    for (int next : graph[node])
        if (!visited[next])
            dfs(next, graph, visited);
}

18. Recursion, Backtracking, Greedy, and Dynamic Programming

Recursion

long long factorial(int n) {
    if (n <= 1) return 1;
    return 1LL * n * factorial(n - 1);
}

Backtracking: generate subsets

void generateSubsets(int index, const vector<int>& a,
                     vector<int>& current) {
    if (index == (int)a.size()) {
        for (int x : current) cout << x << ' ';
        cout << "\n";
        return;
    }

    generateSubsets(index + 1, a, current);

    current.push_back(a[index]);
    generateSubsets(index + 1, a, current);
    current.pop_back();
}

Dynamic programming: Fibonacci

long long fibonacci(int n) {
    vector<long long> dp(n + 1);
    if (n >= 1) dp[1] = 1;

    for (int i = 2; i <= n; i++)
        dp[i] = dp[i - 1] + dp[i - 2];

    return dp[n];
}

Greedy idea

Choose the locally best option at each step. Common examples include activity selection, interval scheduling, Kruskal, Prim, and Huffman coding.

    window += a[i];

21. Heaps, Disjoint Set Union, Shortest Paths, and MST

Disjoint Set Union

class DSU {
    vector<int> parent, size;

public:
    explicit DSU(int n) : parent(n), size(n, 1) {
        iota(parent.begin(), parent.end(), 0);
    }

    int find(int x) {
        if (parent[x] == x) return x;
        return parent[x] = find(parent[x]);
    }

    bool unite(int a, int b) {
        a = find(a);
        b = find(b);
        if (a == b) return false;

        if (size[a] < size[b]) swap(a, b);
        parent[b] = a;
        size[a] += size[b];
        return true;
    }
};

Dijkstra

vector<long long> dijkstra(
    int source,
    const vector<vector<pair<int,int>>>& graph) {

    const long long INF = 4e18;
    vector<long long> dist(graph.size(), INF);

    priority_queue<
        pair<long long,int>,
        vector<pair<long long,int>>,
        greater<pair<long long,int>>
    > pq;

    dist[source] = 0;
    pq.push({0, source});

    while (!pq.empty()) {
        auto [distance, node] = pq.top();
        pq.pop();

        if (distance != dist[node]) continue;

        for (auto [next, weight] : graph[node]) {
            if (dist[next] > distance + weight) {
                dist[next] = distance + weight;
                pq.push({dist[next], next});
            }
        }
    }
    return dist;
}
