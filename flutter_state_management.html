1 · ChangeNotifier — The Model/Controller
task_provider.dart
import 'package:flutter/foundation.dart';

class TaskProvider extends ChangeNotifier {
  List<TaskModel> _tasks = [];
  bool _isLoading = false;
  String? _error;

  // ── Getters (read-only outside) ──
  List<TaskModel> get tasks   => _tasks;
  bool             get loading => _isLoading;
  String?          get error   => _error;

  final _repo = TaskRepository();

  // ── Fetch all tasks ──
  Future<void> fetchTasks() async {
    _isLoading = true;
    _error = null;
    notifyListeners(); // rebuild UI → show spinner

    final result = await _repo.getAllTasks();
    if (result != null) {
      _tasks = result;
    } else {
      _error = 'Failed to load tasks';
    }
    _isLoading = false;
    notifyListeners(); // rebuild UI → show data
  }

  // ── Add a task ──
  Future<void> addTask(String title, String desc) async {
    final ok = await _repo.createTask(title, desc);
    if (ok) await fetchTasks();
  }

  // ── Delete a task ──
  Future<void> deleteTask(String id) async {
    final ok = await _repo.deleteTask(id);
    if (ok) {
      _tasks.removeWhere((t) => t.sId == id);
      notifyListeners();
    }
  }

  // ── Update task status ──
  Future<void> updateStatus(String id, String status) async {
    await _repo.updateStatus(id, status);
    await fetchTasks();
  }
}

class TaskScreen extends StatelessWidget {
  @override
  Widget build(BuildContext context) {

    // .watch → listen + rebuild on every change
    final provider = context.watch<TaskProvider>();

    // .read  → call once (in onPressed, initState)
    // context.read<TaskProvider>().fetchTasks();

4 · Consumer — Rebuild Only Part of UI
consumer_example.dart
class HomePage extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Home')),
      // AppBar does NOT rebuild when provider changes
      body: Consumer<TaskProvider>(
        // Only this subtree rebuilds
        builder: (context, provider, child) {
          if (provider.loading)
            return const Center(
              child: CircularProgressIndicator());
          return ListView.builder(
            itemCount: provider.tasks.length,
            itemBuilder: (_, i) =>
              Text(provider.tasks[i].title ?? ''),
          );
        },
      ),
    );
  }
}

// ── initState: fetch data on screen open ──
@override
void initState() {
  super.initState();
  // use Future.microtask to avoid build-phase call
  Future.microtask(() =>
    context.read<TaskProvider>().fetchTasks());
}
💡
Rule: Use context.watch inside build(). Use context.read inside callbacks like onPressed or initState. Never call watch inside a button handler.
5 · Real Example — Auth Provider (Login/Logout)
auth_provider.dart
class AuthProvider extends ChangeNotifier {
  UserModel? _user;
  bool _loading = false;
  String? _error;

  UserModel? get user    => _user;
  bool       get loading => _loading;
  bool       get isLoggedIn => _user != null;
  String?    get error   => _error;

  Future<bool> login(String email, String pass) async {
    _loading = true; _error = null; notifyListeners();
    final user = await AuthRepository().login(email, pass);
    if (user != null) {
      _user = user;
    } else {
      _error = 'Invalid credentials';
    }
    _loading = false; notifyListeners();
    return user != null;
  }

  void logout() {
    _user = null;
    AuthStorage.logOut();
    notifyListeners();
  }
}

// ── In SignIn screen ──
Future<void> _signIn() async {
  final ok = await context.read<AuthProvider>()
    .login(_emailCtrl.text, _passCtrl.text);
  if (ok) Navigator.pushReplacementNamed(context, '/home');
}
02

Riverpod — Explained with Real Examples

What is Riverpod?
Riverpod is the evolution of Provider by the same author. Key improvements: no BuildContext needed to read state, providers are compile-time safe (no runtime "Provider not found" crashes), and providers can be accessed from anywhere — including outside widgets (services, repositories).
Setup
pubspec.yaml
flutter_riverpod: ^2.5.1
main.dart
// Wrap entire app in ProviderScope — done once, that's it.
void main() {
  runApp(const ProviderScope(child: MyApp()));
}
Unlike Provider, there's no MultiProvider setup needed. Each provider is declared globally at the top of its file. ProviderScope is the single entry point — add it once to main().
Provider Types — Which to Use When
provider_types.dart
// ── Provider: read-only static value ──
final greetingProvider = Provider<String>((ref) {
  return 'Hello, Flutter!';
});

// ── StateProvider: simple mutable value (counter, bool) ──
final counterProvider = StateProvider<int>((ref) => 0);

// ── FutureProvider: wraps an async/API call ──
final tasksProvider = FutureProvider<List<TaskModel>>((ref) async {
  return await TaskRepository().getAllTasks() ?? [];
});

// ── StreamProvider: wraps a live stream ──
final messagesProvider = StreamProvider<List<Message>>((ref) {
  return ChatService().messagesStream();
});

// ── NotifierProvider: replaces ChangeNotifier (Riverpod 2.x) ──
final taskNotifierProvider =
  AsyncNotifierProvider<TaskNotifier, List<TaskModel>>(
    TaskNotifier.new);
Think of it this way: StateProvider → simple value (like a toggle). FutureProvider → one-time API fetch. StreamProvider → live/real-time data. AsyncNotifierProvider → full CRUD logic with async operations.
StateProvider — Simple State (Filter / Toggle)
filter_provider.dart
// ── Declare globally (outside any class) ──
final selectedStatusProvider =
  StateProvider<String>((ref) => 'New');

// ── Widget must extend ConsumerWidget (not StatelessWidget) ──
class FilterBar extends ConsumerWidget {
  const FilterBar({super.key});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    // ref.watch → subscribe to changes (rebuilds widget)
    final selected = ref.watch(selectedStatusProvider);

    return Row(children:
      ['New', 'InProgress', 'Completed'].map((s) =>
        ChoiceChip(
          label:    Text(s),
          selected: selected == s,
          onSelected: (_) =>
            // ref.read(...).notifier.state = value to update
            ref.read(selectedStatusProvider.notifier).state = s,
        )
      ).toList(),
    );
  }
}
The key difference from Provider: use ConsumerWidget instead of StatelessWidget, and access state through ref instead of context. ref.watch = listen (rebuild). ref.read = one-time read (in callbacks).
AsyncNotifierProvider — Real CRUD Example (Task Manager)
AsyncNotifier is the Riverpod 2.x way to handle async state (API calls). It replaces the old StateNotifier. It automatically handles loading / error / data states using AsyncValue — no manual boolean flags needed.
task_notifier.dart
// Step 1: Declare the provider at the top of the file
final taskNotifierProvider =
  AsyncNotifierProvider<TaskNotifier, List<TaskModel>>(
    TaskNotifier.new);

// Step 2: Write the Notifier class
class TaskNotifier extends AsyncNotifier<List<TaskModel>> {
  final _repo = TaskRepository();

  // build() = called once on init. Return the initial data.
  @override
  Future<List<TaskModel>> build() async {
    return await _repo.getAllTasks() ?? [];
  }

  // ── Create task then refresh ──
  Future<void> addTask(String title, String desc) async {
    // state = AsyncLoading() shows spinner automatically
    state = const AsyncLoading();
    await _repo.createTask(title, desc);
    // ref.invalidateSelf() re-calls build() to refresh list
    ref.invalidateSelf();
  }

  // ── Delete: optimistic update (instant UI, no spinner) ──
  Future<void> deleteTask(String id) async {
    final current = state.value ?? [];
    // Update UI instantly before API responds
    state = AsyncData(
      current.where((t) => t.sId != id).toList());
    // Then call API in background
    await _repo.deleteTask(id);
  }
}
Reading AsyncValue in the UI — when/map pattern
AsyncValue has three states: loading, error, data. Use .when() to handle all three cleanly in one place — no need for manual if (_isLoading) checks.
task_screen.dart
class TaskScreen extends ConsumerWidget {
  const TaskScreen({super.key});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    // Watch the provider → AsyncValue<List<TaskModel>>
    final asyncTasks = ref.watch(taskNotifierProvider);

    return Scaffold(
      appBar: AppBar(title: const Text('My Tasks')),
      body: asyncTasks.when(
        // loading: show spinner
        loading: () => const Center(
          child: CircularProgressIndicator()),

        // error: show error message + retry
        error: (err, stack) => Center(
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              Text('Error: $err'),
              ElevatedButton(
                onPressed: () =>
                  // invalidate = force re-fetch
                  ref.invalidate(taskNotifierProvider),
                child: const Text('Retry'),
              ),
            ],
          )),

        // data: render the list
        data: (tasks) => ListView.builder(
          itemCount:   tasks.length,
          itemBuilder: (ctx, i) => ListTile(
            title:    Text(tasks[i].title ?? ''),
            subtitle: Text(tasks[i].status ?? ''),
            trailing: IconButton(
              icon: const Icon(Icons.delete, color: Colors.red),
              onPressed: () => ref
                .read(taskNotifierProvider.notifier)
                .deleteTask(tasks[i].sId!),
            ),
          ),
        ),
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: () => _showAddDialog(context, ref),
        child: const Icon(Icons.add),
      ),
    );
  }

  void _showAddDialog(BuildContext ctx, WidgetRef ref) {
    final ctrl = TextEditingController();
    showDialog(context: ctx, builder: (_) => AlertDialog(
      title: const Text('New Task'),
      content: TextField(controller: ctrl,
        decoration: const InputDecoration(hintText: 'Title')),
      actions: [
        ElevatedButton(
          onPressed: () {
            ref.read(taskNotifierProvider.notifier)
              .addTask(ctrl.text, '');
            Navigator.pop(ctx);
          },
          child: const Text('Add'),
        ),
      ],
    ));
  }
}
Provider Dependency — ref.watch inside a Provider
Riverpod providers can depend on each other using ref.watch inside a provider definition. When the parent provider changes, the dependent provider automatically rebuilds — no manual wiring needed.
filtered_tasks.dart
final selectedStatusProvider =
  StateProvider<String>((ref) => 'New');

// This provider DEPENDS on both tasks + selected filter
final filteredTasksProvider =
  Provider<List<TaskModel>>((ref) {
    final status = ref.watch(selectedStatusProvider);
    // Watch the task list (AsyncValue)
    final asyncTasks = ref.watch(taskNotifierProvider);
    return asyncTasks.value
      ?.where((t) => t.status == status)
      .toList() ?? [];
  });

// In widget: just watch filteredTasksProvider
// It auto-updates when filter OR task list changes
✅
No context needed — you can also read Riverpod providers inside repositories, services, and non-widget code by passing ref as a parameter.

BLoC & Cubit — Explained with Real Examples

What is BLoC?
BLoC stands for Business Logic Component. The pattern separates UI from business logic by using Streams. The UI sends Events → BLoC processes them → emits new States → UI rebuilds. Cubit is a simpler version of BLoC with no Events — you call methods directly instead.
BLoC vs Cubit — When to use which
CubitBLoC
ComplexitySimple state (counter, toggle, form)Complex flows with multiple event types
TriggerCall methods directlyAdd events to the bloc
BoilerplateLow — just methods + statesHigher — needs Event classes too
TraceabilityLess traceableFull event log for debugging
Start with Cubit. Upgrade to BLoC only when you need event history, event transformers (debounce, throttle), or multiple triggers for the same state change.

Cubit — Real Example: Task List (Simple + Clean)
A Cubit has two parts: State classes (what the UI displays) and the Cubit class (the logic that emits those states). The UI listens with BlocBuilder and rebuilds when state changes.
UI calls
cubit.method() → Cubit processes
(API call etc.) → emit(NewState) → BlocBuilder
rebuilds UI
task_state.dart ← Step 1: Define States
// Each possible UI state is its own class
abstract class TaskState {}

// App just started — nothing loaded yet
class TaskInitial extends TaskState {}

// API call in progress — show spinner
class TaskLoading extends TaskState {}

// Data arrived — render the list
class TaskLoaded extends TaskState {
  final List<TaskModel> tasks;
  const TaskLoaded(this.tasks);
}

// Something went wrong — show error
class TaskError extends TaskState {
  final String message;
  const TaskError(this.message);
}
task_cubit.dart ← Step 2: Write the Cubit
import 'package:flutter_bloc/flutter_bloc.dart';

// Cubit<State> — generic type is the state class above
class TaskCubit extends Cubit<TaskState> {
  final _repo = TaskRepository();

  // Initial state passed to super constructor
  TaskCubit() : super(TaskInitial());

  // ── Fetch: Loading → Loaded or Error ──
  Future<void> fetchTasks() async {
    emit(TaskLoading());         // tell UI: show spinner
    final tasks = await _repo.getAllTasks();
    if (tasks != null) {
      emit(TaskLoaded(tasks));   // success → render list
    } else {
      emit(const TaskError('Could not load tasks'));
    }
  }

  // ── Add: optimistic + re-fetch ──
  Future<void> addTask(String title, String desc) async {
    await _repo.createTask(title, desc);
    await fetchTasks(); // refresh the list
  }

  // ── Delete: remove locally for instant feedback ──
  Future<void> deleteTask(String id) async {
    if (state is TaskLoaded) {
      final current = (state as TaskLoaded).tasks;
      // Emit updated list immediately (no spinner)
      emit(TaskLoaded(
        current.where((t) => t.sId != id).toList()));
      await _repo.deleteTask(id);
    }
  }
}

class TaskScreen extends StatelessWidget {
  const TaskScreen({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Tasks')),
      body: BlocBuilder<TaskCubit, TaskState>(
        // builder fires every time cubit emits a new state
        builder: (context, state) {
          if (state is TaskInitial) {
            return const Center(child: Text('Press refresh'));
          }
          if (state is TaskLoading) {
            return const Center(
              child: CircularProgressIndicator());
          }
          if (state is TaskError) {
            return Center(child: Text(state.message));
          }
          // state is TaskLoaded — render data
          final tasks = (state as TaskLoaded).tasks;
          return ListView.builder(
            itemCount:   tasks.length,
            itemBuilder: (ctx, i) => ListTile(
              title:    Text(tasks[i].title ?? ''),
              subtitle: Text(tasks[i].status ?? ''),
              trailing: IconButton(
                icon: const Icon(Icons.delete, color: Colors.red),
                onPressed: () => context.read<TaskCubit>()
                  .deleteTask(tasks[i].sId!),
              ),
            ),
          );
        },
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: () =>
          context.read<TaskCubit>().fetchTasks(),
        child: const Icon(Icons.refresh),
      ),
    );
  }
}
BlocListener & BlocConsumer — Side Effects (Snackbar, Navigate)
BlocBuilder rebuilds the UI. BlocListener reacts to state changes for side effects like showing a snackbar, navigating, or playing a sound — it does NOT rebuild the widget tree. BlocConsumer combines both.
bloc_listener.dart
// ── BlocListener: side effects only ──
BlocListener<TaskCubit, TaskState>(
  listenWhen: (prev, curr) =>
    curr is TaskError, // only fire on error
  listener: (context, state) {
    if (state is TaskError) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text(state.message),
          backgroundColor: Colors.red));
    }
  },
  child: const TaskList(),
)

// ── BlocConsumer: build UI + listen for side effects ──
BlocConsumer<AuthCubit, AuthState>(
  listener: (ctx, state) {
    if (state is AuthLoggedIn) {
      // Navigate after login
      Navigator.pushReplacementNamed(ctx, '/home');
    }
  },
  builder: (ctx, state) {
    if (state is AuthLoading)
      return const CircularProgressIndicator();
    return const LoginForm();
  },
)
Full BLoC Pattern — With Events (Auth Flow)
Full BLoC adds an Events layer. Instead of calling cubit methods, the UI adds Events to the bloc. The bloc handles them with on<EventType> handlers. This makes every action traceable and testable — you can log, replay, or debounce events.
UI add(LoginEvent) → on<LoginEvent>
handler fires → emit(AuthLoaded) → BlocBuilder
rebuilds
auth_event.dart
// Events = what the user/system triggers
abstract class AuthEvent {}

class LoginRequested extends AuthEvent {
  final String email;
  final String password;
  LoginRequested({required this.email, required this.password});
}

class LogoutRequested extends AuthEvent {}
class CheckAuthStatus extends AuthEvent {}
auth_state.dart
abstract class AuthState {}
class AuthInitial   extends AuthState {}
class AuthLoading   extends AuthState {}
class AuthLoggedIn  extends AuthState {
  final UserModel user;
  AuthLoggedIn(this.user);
}
class AuthLoggedOut extends AuthState {}
class AuthError     extends AuthState {
  final String message;
  AuthError(this.message);
}
auth_bloc.dart
class AuthBloc extends Bloc<AuthEvent, AuthState> {
  final _repo = AuthRepository();

  AuthBloc() : super(AuthInitial()) {
    // Register handlers for each event type
    on<LoginRequested>(_onLogin);
    on<LogoutRequested>(_onLogout);
    on<CheckAuthStatus>(_onCheckAuth);
  }

  Future<void> _onLogin(
    LoginRequested event,
    Emitter<AuthState> emit,
  ) async {
    emit(AuthLoading());
    final user = await _repo.login(event.email, event.password);
    if (user != null) {
      emit(AuthLoggedIn(user));
    } else {
      emit(AuthError('Wrong email or password'));
    }
  }

  Future<void> _onLogout(
    LogoutRequested event,
    Emitter<AuthState> emit,
  ) async {
    await AuthStorage.logOut();
    emit(AuthLoggedOut());
  }

  Future<void> _onCheckAuth(
    CheckAuthStatus event,
    Emitter<AuthState> emit,
  ) async {
    final token = await AuthStorage.getToken();
    if (token != null) {
      // token exists → already logged in
      emit(AuthLoggedIn(await AuthStorage.getUser()));
    } else {
      emit(AuthLoggedOut());
    }
  }
}

          padding: const EdgeInsets.all(24),
          child: Column(children: [
            TextField(controller: _emailCtrl,
              decoration: const InputDecoration(labelText: 'Email')),
            const SizedBox(height: 12),
            TextField(controller: _passCtrl,
              obscureText: true,
