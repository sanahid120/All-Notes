6. Null Safety

A non-nullable variable cannot contain null. Add ? to make a type nullable.

void main() {
  String name = 'Nahid';
  String? middleName;

  print(name);
  print(middleName); // null
}

Null-aware operators

void main() {
  String? username;

  String displayName = username ?? 'Guest';
  print(displayName);

  username ??= 'Default User';
  print(username);

  String? text = 'dart';
  print(text?.toUpperCase());

  String definitelyNotNull = text!;
  print(definitelyNotNull);
}

late keyword

class UserService {
  late String token;

  void login() {
    token = 'abc123';
  }

  void showToken() {
    print(token);
  }
}

20. FutureBuilder Flutter-specific

FutureBuilder is a Flutter widget that rebuilds the UI according to the state of a Future.

import 'package:flutter/material.dart';

class UserPage extends StatefulWidget {
  const UserPage({super.key});

  @override
  State<UserPage> createState() => _UserPageState();
}

class _UserPageState extends State<UserPage> {
  late final Future<String> _userFuture;

  @override
  void initState() {
    super.initState();
    _userFuture = fetchUser();
  }

  Future<String> fetchUser() async {
    await Future.delayed(const Duration(seconds: 2));
    return 'Nahid';
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('FutureBuilder Example')),
      body: FutureBuilder<String>(
        future: _userFuture,
        builder: (context, snapshot) {
          if (snapshot.connectionState == ConnectionState.waiting) {
            return const Center(child: CircularProgressIndicator());
          }

          if (snapshot.hasError) {
            return Center(child: Text('Error: ${snapshot.error}'));
          }

          if (!snapshot.hasData) {
            return const Center(child: Text('No data found'));
          }

          return Center(child: Text('Welcome, ${snapshot.data}'));
        },
      ),
    );
  }
}
Best practice: Create the Future in initState() or store it in a provider/view model. Avoid creating a new Future directly inside build() when repeated calls are not intended.

22. Linked List

A singly linked list contains nodes. Each node stores data and a reference to the next node.

class Node<T> {
  T value;
  Node<T>? next;

  Node(this.value);
}

class LinkedList<T> {
  Node<T>? head;

  void addFirst(T value) {
    final newNode = Node(value);
    newNode.next = head;
    head = newNode;
  }

  void addLast(T value) {
    final newNode = Node(value);

    if (head == null) {
      head = newNode;
      return;
    }

    Node<T> current = head!;
    while (current.next != null) {
      current = current.next!;
    }
    current.next = newNode;
  }

  bool contains(T value) {
    Node<T>? current = head;

    while (current != null) {
      if (current.value == value) return true;
      current = current.next;
    }
    return false;
  }

  bool remove(T value) {
    if (head == null) return false;

    if (head!.value == value) {
      head = head!.next;
      return true;
    }

    Node<T> current = head!;
    while (current.next != null) {
      if (current.next!.value == value) {
        current.next = current.next!.next;
        return true;
      }
      current = current.next!;
    }
    return false;
  }

  void printList() {
    Node<T>? current = head;
    final values = <T>[];

    while (current != null) {
      values.add(current.value);
      current = current.next;
    }

    print(values.join(' -> '));
  }
}

void main() {
  final list = LinkedList<int>();
  list.addLast(10);
  list.addLast(20);
  list.addFirst(5);
  list.printList();

  print(list.contains(20));
  list.remove(10);
  list.printList();
}

23. Stack and Queue

Stack using List

class Stack<T> {
  final List<T> _items = [];

  void push(T item) => _items.add(item);

  T pop() {
    if (_items.isEmpty) {
      throw StateError('Stack is empty');
    }
    return _items.removeLast();
  }

  T get peek {
    if (_items.isEmpty) {
      throw StateError('Stack is empty');
    }
    return _items.last;
  }

  bool get isEmpty => _items.isEmpty;
  int get length => _items.length;
}

void main() {
  final stack = Stack<String>();
  stack.push('A');
  stack.push('B');
  stack.push('C');

  print(stack.peek);
  print(stack.pop());
  print(stack.pop());
}

24. Searching Algorithms

Linear search — O(n)

int linearSearch(List<int> numbers, int target) {
  for (int i = 0; i < numbers.length; i++) {
    if (numbers[i] == target) {
      return i;
    }
  }
  return -1;
}

void main() {
  print(linearSearch([5, 8, 2, 10], 2)); // 2
}

Binary search — O(log n)

Binary search requires a sorted list.

int binarySearch(List<int> numbers, int target) {
  int left = 0;
  int right = numbers.length - 1;

  while (left <= right) {
    final middle = left + (right - left) ~/ 2;

    if (numbers[middle] == target) {
      return middle;
    } else if (numbers[middle] < target) {
      left = middle + 1;
    } else {
      right = middle - 1;
    }
  }

  return -1;
}

void main() {
  print(binarySearch([1, 3, 5, 7, 9, 11], 7)); // 3
}

Bubble sort — O(n²)

List<int> bubbleSort(List<int> input) {
  final numbers = List<int>.from(input);

  for (int i = 0; i < numbers.length - 1; i++) {
    bool swapped = false;

    for (int j = 0; j < numbers.length - i - 1; j++) {
      if (numbers[j] > numbers[j + 1]) {
        final temp = numbers[j];
        numbers[j] = numbers[j + 1];
        numbers[j + 1] = temp;
        swapped = true;
      }
    }

    if (!swapped) break;
  }

  return numbers;
}

Selection sort — O(n²)

List<int> selectionSort(List<int> input) {
  final numbers = List<int>.from(input);

  for (int i = 0; i < numbers.length - 1; i++) {
    int minIndex = i;

    for (int j = i + 1; j < numbers.length; j++) {
      if (numbers[j] < numbers[minIndex]) {
        minIndex = j;
      }
    }

    final temp = numbers[i];
    numbers[i] = numbers[minIndex];
    numbers[minIndex] = temp;
  }

  return numbers;
}

Insertion sort — O(n²)

List<int> insertionSort(List<int> input) {
  final numbers = List<int>.from(input);

  for (int i = 1; i < numbers.length; i++) {
    final key = numbers[i];
    int j = i - 1;

    while (j >= 0 && numbers[j] > key) {
      numbers[j + 1] = numbers[j];
      j--;
    }

    numbers[j + 1] = key;
  }

  return numbers;
}

Merge sort — O(n log n)

List<int> mergeSort(List<int> numbers) {
  if (numbers.length <= 1) return numbers;

  final middle = numbers.length ~/ 2;
  final left = mergeSort(numbers.sublist(0, middle));
  final right = mergeSort(numbers.sublist(middle));

  return merge(left, right);
}

List<int> merge(List<int> left, List<int> right) {
  final result = <int>[];
  int i = 0;
  int j = 0;

  while (i < left.length && j < right.length) {
    if (left[i] <= right[j]) {
      result.add(left[i++]);
    } else {
      result.add(right[j++]);
    }
  }

  result.addAll(left.sublist(i));
  result.addAll(right.sublist(j));
  return result;
}

27. Complete Mini Project: Product Manager

This console project combines input, parsing, enums, named parameters, null safety, collections, OOP, JSON, factory constructors, and sorting.

import 'dart:convert';
import 'dart:io';

enum ProductCategory {
  phone,
  laptop,
  accessory,
}

class Product {
  final int id;
  final String name;
  final double price;
  final ProductCategory category;
  final List<String> images;
  final String? description;

  const Product({
    required this.id,
    required this.name,
    required this.price,
    required this.category,
    required this.images,
    this.description,
  });

  factory Product.fromJson(Map<String, dynamic> json) {
    return Product(
      id: json['id'] as int,
      name: json['name'] as String,
      price: (json['price'] as num).toDouble(),
      category: ProductCategory.values.firstWhere(
        (item) => item.name == json['category'],
        orElse: () => ProductCategory.accessory,
      ),
      images: List<String>.from(json['images'] ?? const []),
      description: json['description'] as String?,
    );
  }

  Map<String, dynamic> toJson() {
    return {
      'id': id,
      'name': name,
      'price': price,
      'category': category.name,
      'images': images,
      'description': description,
    };
  }

  Product copyWith({
    int? id,
    String? name,
    double? price,
    ProductCategory? category,
    List<String>? images,
    String? description,
  }) {
    return Product(
      id: id ?? this.id,
      name: name ?? this.name,
      price: price ?? this.price,
      category: category ?? this.category,
      images: images ?? this.images,
      description: description ?? this.description,
    );
  }

  @override
  String toString() {
    return '$id | $name | \$$price | ${category.name}';
  }
}

class ProductRepository {
  final List<Product> _products = [];

  List<Product> get products => List.unmodifiable(_products);

  void add(Product product) {
    _products.add(product);
  }

  bool deleteById(int id) {
    final previousLength = _products.length;
    _products.removeWhere((product) => product.id == id);
    return _products.length != previousLength;
  }

  Product? findById(int id) {
    for (final product in _products) {
      if (product.id == id) return product;
    }
    return null;
  }

  List<Product> search(String query) {
    final normalized = query.toLowerCase();
    return _products
        .where((product) => product.name.toLowerCase().contains(normalized))
        .toList();
  }

  void sortByPrice({bool ascending = true}) {
    _products.sort(
      (a, b) => ascending
          ? a.price.compareTo(b.price)
          : b.price.compareTo(a.price),
    );
  }

  String exportJson() {
    return jsonEncode(_products.map((product) => product.toJson()).toList());
  }
}

void main() {
  final repository = ProductRepository();

  repository.add(
    const Product(
      id: 1,

  );

      category: ProductCategory.laptop,
      images: ['laptop.jpg'],

  stdout.write('Search product: ');

  final results = repository.search(query);
  if (results.isEmpty) {
    print('No products found.');
  } else {
    for (final product in results) {
      print(product);
    }
  }

  repository.sortByPrice();
  print('\nSorted products:');
  repository.products.forEach(print);

  print('\nJSON output:');
  print(repository.exportJson());
}

28. Modern Dart: Records, Patterns, and Switch Expressions

Records let a function return several values without creating a one-off class. Pattern matching makes those values easy to inspect.

({String name, int age}) profile() => (name: 'Mina', age: 24);

void main() {
  final (name: person, age: years) = profile();

  final label = switch (years) {
    < 13 => 'child',
    < 20 => 'teenager',
    _ => 'adult',
  };
  print(label);
}
Use records for lightweight grouped values. Prefer a named class when the value needs behavior, validation, or a stable public API.

30. Isolates and When to Use Them

Dart isolates have separate memory and communicate by messages. Use Isolate.run for CPU-heavy work that would otherwise block a Flutter frame; ordinary asynchronous I/O does not usually need an isolate.

import 'dart:isolate';

Future<int> calculateInBackground(List<int> values) {
  return Isolate.run(() => values.fold(0, (sum, value) => sum + value));
}
Keep isolate inputs and outputs simple and transferable. For very large jobs or persistent workers, learn ports and isolate lifecycle management.
